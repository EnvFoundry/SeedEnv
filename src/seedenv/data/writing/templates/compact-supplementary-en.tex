\newcommand{\natureplotpanel}[3]{%
  \begin{tikzpicture}[x=1mm,y=1mm]
    \draw[ResearchBlue!35,line width=0.35pt] (0,0) rectangle (#2,#3);
    \node[anchor=north west,font=\sffamily\bfseries\fontsize{8}{8}\selectfont,
      text=ResearchBlueDark]
      at (1,#3-1) {#1};
    \draw[black!70,line width=0.35pt] (8,7) -- (#2-6,7);
    \draw[black!70,line width=0.35pt] (8,7) -- (8,#3-7);
    \draw[NatureSkyBlue,line width=0.8pt] plot[smooth] coordinates
      {(9,9) (16,12) (24,22) (34,27) (#2-8,#3-8)};
    \draw[NatureOrange,line width=0.8pt] plot[smooth] coordinates
      {(9,10) (16,11) (24,16) (34,22) (#2-8,#3-11)};
    \node[font=\sffamily\fontsize{6}{6}\selectfont] at (#2/2,3.2) {Input};
    \node[rotate=90,font=\sffamily\fontsize{6}{6}\selectfont] at (3.2,#3/2)
      {Response};
  \end{tikzpicture}}

\newcommand{\natureimagepanel}[4]{%
  \begin{tikzpicture}[x=1mm,y=1mm]
    \shade[left color=black!12,right color=black!3] (0,0) rectangle (#2,#3);
    \draw[ResearchBlue!35,line width=0.35pt] (0,0) rectangle (#2,#3);
    \node[anchor=north west,font=\sffamily\bfseries\fontsize{8}{8}\selectfont,
      text=ResearchBlueDark]
      at (1,#3-1) {#1};
    \foreach \rx/\ry in {0.25/0.32,0.42/0.62,0.64/0.42,0.82/0.70,0.58/0.80} {
      \fill[black!27] ({#2*\rx},{#3*\ry}) circle (2.2);
    }
    \draw[black,line width=0.9pt] (#2-19,5) -- (#2-6,5);
    \node[black,font=\sffamily\fontsize{6}{6}\selectfont,anchor=south]
      at (#2-12.5,5.6) {50 \textmu m};
    \node[font=\sffamily\fontsize{6}{6}\selectfont,fill=white,inner sep=1pt]
      at (#2/2,#3-5) {#4};
  \end{tikzpicture}}
